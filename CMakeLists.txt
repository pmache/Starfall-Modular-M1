cmake_minimum_required(VERSION 3.16)
project(StarfallModular VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(STARFALL_BUILD_GAME "Build the raylib graphical executable" ON)
option(STARFALL_SANITIZE "Enable address/undefined sanitizers (GCC/Clang)" OFF)
include(CTest)
add_library(starfall_core
    src/content/json.cpp src/content/catalog.cpp src/gameplay/campaign.cpp
    src/gameplay/game.cpp src/gameplay/simulate.cpp src/runtime/options.cpp src/runtime/paths.cpp)
if(MSVC)
    target_compile_options(starfall_core PUBLIC /W4 /WX /utf-8)
else()
    target_compile_options(starfall_core PUBLIC -Wall -Wextra -Werror -D_GLIBCXX_ASSERTIONS)
    if(STARFALL_SANITIZE)
        target_compile_options(starfall_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(starfall_core PUBLIC -fsanitize=address,undefined)
    endif()
endif()
if(WIN32)
    target_link_libraries(starfall_core PUBLIC shell32 user32)
endif()
add_library(starfall_regression tests/regression.cpp)
target_link_libraries(starfall_regression PUBLIC starfall_core)
add_executable(starfall-content src/content/tool_main.cpp)
target_link_libraries(starfall-content PRIVATE starfall_regression)
# Run tools from any CWD; content always lives beside the binary by default.
add_custom_command(TARGET starfall-content POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/content" "$<TARGET_FILE_DIR:starfall-content>/content")
if(STARFALL_BUILD_GAME)
    find_package(raylib 5.5 QUIET)
    if(NOT raylib_FOUND)
        find_path(RAYLIB_INCLUDE_DIR raylib.h)
        find_library(RAYLIB_LIBRARY NAMES raylib)
        if(NOT RAYLIB_INCLUDE_DIR OR NOT RAYLIB_LIBRARY)
            message(FATAL_ERROR "Install raylib 5.5 or use -DSTARFALL_BUILD_GAME=OFF for core tests")
        endif()
        add_library(raylib UNKNOWN IMPORTED)
        set_target_properties(raylib PROPERTIES IMPORTED_LOCATION "${RAYLIB_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${RAYLIB_INCLUDE_DIR}")
    endif()
    add_executable(starfall WIN32 main.cpp src/runtime/renderer.cpp)
    target_link_libraries(starfall PRIVATE starfall_regression raylib)
    if(WIN32)
        target_link_libraries(starfall PRIVATE opengl32 gdi32 winmm)
    endif()
    add_custom_command(TARGET starfall POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_SOURCE_DIR}/content" "$<TARGET_FILE_DIR:starfall>/content")
endif()
if(BUILD_TESTING)
    add_executable(starfall-tests tests/test_main.cpp)
    target_link_libraries(starfall-tests PRIVATE starfall_regression)
    add_test(NAME content_and_campaign COMMAND starfall-tests "${CMAKE_SOURCE_DIR}/content")
    add_test(NAME original_simulation COMMAND starfall-content --self-test)
    add_test(NAME validate_content COMMAND starfall-content --validate-content)
    add_test(NAME campaign_demo COMMAND starfall-content --simulate-campaign --campaign example:two_sectors)
    add_test(NAME sector_preview COMMAND starfall-content --simulate-campaign --sector example:relay)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME no_recompile_and_bad_content COMMAND ${Python3_EXECUTABLE}
            "${CMAKE_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:starfall-content>" "${CMAKE_SOURCE_DIR}/content")
    endif()
    if(STARFALL_BUILD_GAME)
        add_test(NAME full_game_self_test COMMAND starfall --self-test)
    endif()
endif()
